async function fetchNews(cat) { currentCat = cat; const container = document.getElementById('feed-container'); container.innerHTML = '
CHARGEMENT...
'; try { let all = []; for (let url of SOURCES[cat]) { const res = await fetch(`https://api.rss2json.com/v1/api.json?rss_url=${encodeURIComponent(url)}`); const d = await res.json(); if (d.status === 'ok') all = [...all, ...d.items.map(i => ({...i, src: d.feed.title.substring(0,12)}))]; } all.sort((a, b) => new Date(b.pubDate) - new Date(a.pubDate)); container.innerHTML = ''; all.slice(0, 15).forEach(item => { const div = document.createElement('div'); div.className = 'news-item'; div.onclick = (e) => { // Empêche l'ouverture/fermeture de la carte si on clique sur le bouton lien if(e.target.tagName !== 'DIV' || !e.target.onclick) { div.classList.toggle('open'); doVibrate(20); } }; const desc = (item.description || "").replace(/<[^>]*>?/gm, '').substring(0, 160); div.innerHTML = `
${item.src}${new Date(item.pubDate).toLocaleTimeString('fr-FR', {hour:'2-digit', minute:'2-digit'})}
${item.title}
${desc}...

VOIR L'ARTICLE
`; container.appendChild(div); }); } catch (e) { container.innerHTML = '
ERREUR RÉSEAU
'; } }
